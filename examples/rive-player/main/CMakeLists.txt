    set(RIV_ARTBOARD "" CACHE STRING "Named artboard (empty uses the file default)")
    set(asset "${RIV_FILE}")
    set(board_name "${RIV_ARTBOARD}")
    if(NOT asset)
        set(asset "${CMAKE_CURRENT_LIST_DIR}/../../../assets/watch/pocket-watch.riv")
        if(NOT board_name)
            set(board_name "Movement")
        endif()
    endif()
    if(NOT EXISTS "${asset}")
        message(FATAL_ERROR "RIV_FILE must point to an existing .riv file")
    endif()
    target_add_binary_data(${COMPONENT_LIB} "${asset}" BINARY RENAME_TO scene_riv)
    string(REPLACE "\\" "\\\\" board_name "${board_name}")
    string(REPLACE "\"" "\\\"" board_name "${board_name}")
    configure_file(settings.h.in settings.h @ONLY)
    target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
endif()

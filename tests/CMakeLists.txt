function(board_test name source_dir)
    add_executable(test-${name} ${name}.c "${source_dir}/${name}.c")
    target_include_directories(test-${name} PRIVATE "${PROJECT_SOURCE_DIR}/board" "${source_dir}")
    # Assertions in the host fixtures also exercise the C API in release builds.
    target_compile_options(test-${name} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(test-${name} PRIVATE m)
    add_test(NAME ${name} COMMAND test-${name} ${ARGN})
endfunction()

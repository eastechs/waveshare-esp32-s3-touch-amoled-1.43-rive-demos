cmake_minimum_required(VERSION 3.20)
project(waveshare_esp32_s3_touch_amoled_143_rive_demos LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(WITH_RIVE "Test the live Rive thermostat" ON)
set(BOARD_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/deps/waveshare-esp32-s3-touch-amoled-1.43-bsp/board")
if(NOT EXISTS "${BOARD_HEADERS}/display.h")
    message(FATAL_ERROR "Initialize the deps/waveshare-esp32-s3-touch-amoled-1.43-bsp submodule before building the demo tests.")
endif()
if(WITH_RIVE)
    add_subdirectory(deps/rive-thorvg-framebuffer)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
